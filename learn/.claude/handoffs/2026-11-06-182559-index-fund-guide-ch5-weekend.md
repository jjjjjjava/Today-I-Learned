## 从 epub 取正文与图（只用标准库）

```bash
mkdir -p /tmp/ch5 && cd /tmp/ch5
python3 -I - <<'PY'
import zipfile,re
z=zipfile.ZipFile('<仓库路径>/learn/书/指数基金投资指南.epub')
for n in sorted(x for x in z.namelist() if re.match(r'text/part00(3[2-9]|4\d)',x)):
    t=z.read(n).decode()
    t=re.sub(r'<img[^>]*src="\.\./(images/[^"]+)"[^>]*/>',r'[IMG \1]',t)
    t=re.sub(r'<[^>]+>','',t); t=re.sub(r'\n\s*\n+','\n',t)
    open(n.split('/')[1]+'.txt','w').write(t)
PY
```
